function initDragAndDropLogic() {
const cards = document.querySelectorAll('.smart-card[draggable="true"]'); const grid = document.getElementById('device-grid');
cards.forEach(card => {
card.addEventListener('dragstart', () => { window.isDragging = true; card.classList.add('dragging'); });
card.addEventListener('dragend', () => { card.classList.remove('dragging'); salvaNuovoOrdineCloud(); setTimeout(() => window.isDragging = false, 100); });
let touchTimer;
card.addEventListener('touchstart', (e) => { touchTimer = setTimeout(() => { window.isDragging = true; card.classList.add('dragging'); }, 350); });
card.addEventListener('touchmove', (e) => {
if (!card.classList.contains('dragging')) { clearTimeout(touchTimer); return; } e.preventDefault();
const touch = e.touches[0]; const targetEl = document.elementFromPoint(touch.clientX, touch.clientY); const currentCardOver = targetEl ? targetEl.closest('.smart-card[draggable="true"]') : null;
if (currentCardOver && currentCardOver !== card) { const allCards = [...grid.querySelectorAll('.smart-card[draggable="true"]:not(.dragging)')]; const nextEl = allCards.find(c => { const box = c.getBoundingClientRect(); return touch.clientY < box.top + box.height / 2 && touch.clientX < box.left + box.width / 2; }); grid.insertBefore(card, nextEl); }
});
card.addEventListener('touchend', () => { clearTimeout(touchTimer); if (card.classList.contains('dragging')) { card.classList.remove('dragging'); salvaNuovoOrdineCloud(); setTimeout(() => window.isDragging = false, 100); } });
});
grid.addEventListener('dragover', (e) => { e.preventDefault(); const draggingCard = grid.querySelector('.dragging'); if(!draggingCard) return; const siblings = [...grid.querySelectorAll('.smart-card[draggable="true"]:not(.dragging)')]; let nextSibling = siblings.find(sibling => { const box = sibling.getBoundingClientRect(); return e.clientY < box.top + box.height / 2 && e.clientX < box.left + box.width / 2; }); grid.insertBefore(draggingCard, nextSibling); });
}

window.togglePowerHon = async function(id, forceUpdate = false) {
const s = window.stanzeState[id];
if(!s) return;
const oldMode = s.mode;
let newMode = s.mode;
if (!forceUpdate) {
newMode = (s.mode === "off") ? "cool" : "off";
}
// UI Feedback immediato (Optimistic UI)
s.mode = newMode;
window.aggiornaUIStanza(id);
try {
// Chiamata alla cartella "api" di Vercel creata precedentemente
const response = await fetch('/api/hon', {
method: 'POST',
headers: { 'Content-Type': 'application/json' },
body: JSON.stringify({
command: newMode === "off" ? "off" : "on",
temperature: s.temp
})
});
if(!response.ok) {
const result = await response.json();
alert("Errore hOn Server: " + (result.error || "Errore Sconosciuto"));
s.mode = oldMode; // Revert UI se fallisce
window.aggiornaUIStanza(id);
}
} catch(error) {
alert("Errore di rete! Impossibile raggiungere Vercel.");
s.mode = oldMode; // Revert UI se fallisce
window.aggiornaUIStanza(id);
}
};
