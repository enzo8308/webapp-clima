if(clientMQTT.connected) eseguiIscrizioni();

clientMQTT.subscribe(`vincenzo_clima_segreto_7788/${idStanzaURL}/status`);
clientMQTT.subscribe(topicBaseMQTT + "/mode/state");
clientMQTT.subscribe(topicBaseMQTT + "/target_temperature/state");
clientMQTT.subscribe(topicBaseMQTT + "/current_temperature/state");
clientMQTT.subscribe(topicBaseMQTT + "/fan_mode/state");
clientMQTT.subscribe(topicBaseMQTT + "/swing_mode/state");
clientMQTT.subscribe(`vincenzo_clima_segreto_7788/switch/${luceSwitchName}/state`);

clientMQTT.on('message', function (topic, message) {
const payload = message.toString().toLowerCase();
if (topic.endsWith("/status")) {
const badge = document.getElementById('status-badge');
const wifiIcon = document.getElementById('top-wifi-icon');
if (payload === "online") {
isOnline = true; badge.innerText = "ONLINE"; badge.className = "status-badge online";
if(wifiIcon) wifiIcon.style.color = "#4cd964";
} else {
isOnline = false; badge.innerText = "OFFLINE"; badge.className = "status-badge offline";
if(wifiIcon) wifiIcon.style.color = "rgba(255,255,255,0.5)";
}
}
else if (topic.includes("/mode/state")) {
window.syncActiveButtons('.btn-mode', payload);
if (payload === "off") {
climaAcceso = false;
document.body.style.setProperty('--theme-color', '#888888');
document.body.style.setProperty('--theme-bg', '#e2e3e5');

climaAcceso = true;
document.getElementById('base-power-btn').classList.remove('off');
document.getElementById('top-power-checkbox').checked = true;
document.getElementById('lbl-off').className = "";
document.getElementById('lbl-on').className = "active-text";
document.querySelectorAll('.smart-tile.theme-active').forEach(t => t.style.opacity = '1');
let color = '#007bff'; let bg = '#eaf4ff'; let testo = 'Fresco'; let icona = 'fas fa-snowflake';
if (payload === 'heat') { color = '#ff3b30'; bg = '#ffebe9'; testo = 'Caldo'; icona = 'fas fa-sun';}
else if (payload === 'dry') { color = '#5ac8fa'; bg = '#eef9ff'; testo = 'Deumid.'; icona = 'fas fa-tint';}
else if (payload === 'fan_only') { color = '#8e8e93'; bg = '#f2f2f7'; testo = 'Ventola'; icona = 'fas fa-fan';}
else if (payload === 'auto') { color = '#34c759'; bg = '#ebf9ee'; testo = 'Auto'; icona = 'fas fa-magic';}
document.body.style.setProperty('--theme-color', color);
document.body.style.setProperty('--theme-bg', bg);
document.querySelectorAll('.testo-stato').forEach(t => t.innerText = testo);
document.querySelectorAll('.icona-stato').forEach(i => i.className = icona + " icona-stato");

window.aggiornaTermostato(gradiAttuali, false);
}
}
else if (topic.includes("/target_temperature/state")) {
const tempVal = parseFloat(payload);
gradiAttuali = isNaN(tempVal) ? 24 : tempVal;
document.querySelectorAll('.valore-temp').forEach(el => el.innerText = gradiAttuali);
document.querySelectorAll('.valore-temp-testo').forEach(el => el.innerText = gradiAttuali + " °C");
window.aggiornaTermostato(gradiAttuali, !climaAcceso);
}
else if (topic.includes("/current_temperature/state")) {
const currTemp = parseFloat(payload);
document.querySelectorAll('.current-temp-val').forEach(el => el.innerText = isNaN(currTemp) ? "--" : currTemp);
}

document.getElementById('lbl-tile-fan').innerText = lbl;
}
else if (topic.includes("/swing_mode/state")) {
window.syncActiveButtons('.btn-vane', payload);
let lbl = "Ferma";

window.aggiornaTermostato = function(temp, isEmpty) {
const minT = 16; const maxT = 30;
let val = temp; if(val < minT) val = minT; if(val > maxT) val = maxT;
const progress = (val - minT) / (maxT - minT);
const dashOffset = isEmpty ? 660 : 660 - (progress * 660);
const path = document.getElementById('pro-arc-path');
if(path) path.style.strokeDashoffset = dashOffset;
const angle = -135 + (progress * 270);
const thumb = document.getElementById('pro-arc-thumb');
if(thumb) thumb.style.transform = `rotate(${angle}deg)`;
}
window.syncActiveButtons = function(selectorClass, activeValue) {

if(btn.dataset.val === activeValue) btn.classList.add('active');
else btn.classList.remove('active');
});
}
